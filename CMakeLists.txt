cmake_minimum_required(VERSION 3.25)
project(WorldForge VERSION 0.0.1 LANGUAGES C CXX)

option(WF_BUILD_STUDIO "Build Windows Studio" ON)
option(WF_BUILD_TESTS "Build CPU tests" ON)
option(WF_ENABLE_GPU_TESTS "Register GPU smoke test (requires graphics session)" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
set(FETCHCONTENT_BASE_DIR "${CMAKE_SOURCE_DIR}/build/_deps" CACHE PATH "Shared dependency cache")

# Exact upstream commits; no floating branches.
FetchContent_Declare(glm URL https://github.com/g-truc/glm/archive/0af55ccecd98d4e5a8d1fad7de25ba429d60e863.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
set(GLM_BUILD_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(glm)

add_library(wf_editor src/editor/editor_state.cc)
target_include_directories(wf_editor PUBLIC src)
target_link_libraries(wf_editor PUBLIC glm::glm)
target_compile_definitions(wf_editor PUBLIC GLM_FORCE_RADIANS)
if(MSVC)
  target_compile_options(wf_editor PUBLIC /utf-8 /W4 /permissive-)
endif()

if(WF_BUILD_STUDIO)
  if(NOT WIN32)
    message(FATAL_ERROR "Studio currently targets Windows. Other platforms are planned later.")
  endif()
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SDL_REVISION "SDL-3.2.28-7f3ae3d57459" CACHE STRING "Pinned SDL revision" FORCE)
  FetchContent_Declare(SDL3 URL https://github.com/libsdl-org/SDL/archive/7f3ae3d57459e59943a4ecfefc8f6277ec6bf540.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_Declare(imgui URL https://github.com/ocornut/imgui/archive/af84778eeaab75bee0b54420f3157cf3ee00618f.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_Declare(glad URL https://github.com/Dav1dde/glad/archive/73db193f853e2ee079bf3ca8a64aa2eaf6459043.tar.gz SOURCE_SUBDIR cmake DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(SDL3 imgui glad)
  glad_add_library(wf_glad STATIC REPRODUCIBLE API gl:core=4.5)

  add_library(wf_imgui STATIC
    ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl3.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
  target_include_directories(wf_imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
  target_link_libraries(wf_imgui PUBLIC SDL3::SDL3 opengl32)

  add_library(wf_renderer src/render/viewport_renderer.cc)
  target_include_directories(wf_renderer PUBLIC src)
  target_link_libraries(wf_renderer PUBLIC wf_editor PRIVATE wf_glad)
  add_library(wf_platform src/platform/sdl_host.cc)
  target_include_directories(wf_platform PUBLIC src)
  target_link_libraries(wf_platform PUBLIC SDL3::SDL3 PRIVATE wf_glad)
  add_executable(worldforge-studio WIN32
    apps/studio/main.cc src/studio/studio_app.cc src/ui/studio_panels.cc)
  target_link_libraries(worldforge-studio PRIVATE wf_editor wf_renderer wf_platform wf_imgui wf_glad)
  target_compile_definitions(worldforge-studio PRIVATE WF_VERSION="${PROJECT_VERSION}")
  install(TARGETS worldforge-studio RUNTIME DESTINATION .)
  install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
  install(DIRECTORY docs DESTINATION .)
  install(FILES ${SDL3_SOURCE_DIR}/LICENSE.txt DESTINATION licenses RENAME SDL.txt)
  install(FILES ${imgui_SOURCE_DIR}/LICENSE.txt DESTINATION licenses RENAME imgui.txt)
  install(FILES ${glm_SOURCE_DIR}/copying.txt DESTINATION licenses RENAME glm.txt)
  install(FILES ${glad_SOURCE_DIR}/LICENSE DESTINATION licenses RENAME glad.txt)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
  include(InstallRequiredSystemLibraries)
endif()

if(WF_BUILD_TESTS)
  enable_testing()
  add_executable(wf_editor_tests tests/editor_state_test.cc)
  target_link_libraries(wf_editor_tests PRIVATE wf_editor)
  add_test(NAME editor_state COMMAND wf_editor_tests)
  if(WF_BUILD_STUDIO AND WF_ENABLE_GPU_TESTS)
    add_test(NAME studio_gpu_smoke COMMAND worldforge-studio --smoke-test "${CMAKE_BINARY_DIR}/smoke")
    set_tests_properties(studio_gpu_smoke PROPERTIES TIMEOUT 60 LABELS "gpu")
  endif()
endif()
