cmake_minimum_required(VERSION 3.25)
project(WorldForge VERSION 0.0.2 LANGUAGES C CXX)

option(WF_BUILD_STUDIO "Build Windows Studio" ON)
option(WF_BUILD_TESTS "Build CPU tests" ON)
option(WF_ENABLE_GPU_TESTS "Register GPU smoke test (requires graphics session)" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
set(FETCHCONTENT_BASE_DIR "${CMAKE_SOURCE_DIR}/build/_deps" CACHE PATH "Shared dependency cache")

# Exact upstream commits; no floating branches.
FetchContent_Declare(glm URL https://github.com/g-truc/glm/archive/0af55ccecd98d4e5a8d1fad7de25ba429d60e863.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
set(GLM_BUILD_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(glm)

set(TINYGLTF3_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(TINYGLTF3_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(tinygltf URL https://github.com/syoyo/tinygltf/archive/0d9ce1a9561688c6c6e7471430d15a50a0b15f1a.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(tinygltf)
add_library(wf_tinygltf STATIC ${tinygltf_SOURCE_DIR}/tiny_gltf_v3.c)
target_include_directories(wf_tinygltf PUBLIC ${tinygltf_SOURCE_DIR})
set_target_properties(wf_tinygltf PROPERTIES C_STANDARD 11)
# PNG decoder comes from the same exact tinygltf archive; no floating download.
add_library(wf_png STATIC ${tinygltf_SOURCE_DIR}/attic/examples/common/lodepng.cpp)
target_include_directories(wf_png PUBLIC ${tinygltf_SOURCE_DIR}/attic/examples/common)
target_compile_definitions(wf_png PUBLIC LODEPNG_NO_COMPILE_CPP LODEPNG_NO_COMPILE_ENCODER
  LODEPNG_NO_COMPILE_DISK LODEPNG_NO_COMPILE_ANCILLARY_CHUNKS LODEPNG_NO_COMPILE_ALLOCATORS)
add_library(wf_domain src/domain/mesh_asset.cpp src/domain/imported_scene.cpp)
target_include_directories(wf_domain PUBLIC src)
target_link_libraries(wf_domain PUBLIC glm::glm)
target_compile_definitions(wf_domain PUBLIC GLM_FORCE_RADIANS)
add_library(wf_import src/io/glb_importer.cpp src/io/png_decoder.cpp src/io/glb_textures.cpp src/application/import_service.cpp)
target_link_libraries(wf_import PUBLIC wf_domain PRIVATE wf_tinygltf wf_png)
# 导入使用单个可等待的工作线程，CPU 目标不依赖窗口。
# ImportService uses one joined worker; keep the CPU target window-independent.
find_package(Threads REQUIRED)
target_link_libraries(wf_import PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(wf_tinygltf PRIVATE /utf-8)
  target_compile_options(wf_domain PUBLIC /utf-8 /W4 /permissive-)
endif()
add_library(wf_editor src/editor/editor_state.cpp)
target_include_directories(wf_editor PUBLIC src)
target_link_libraries(wf_editor PUBLIC wf_domain)
target_compile_definitions(wf_editor PUBLIC GLM_FORCE_RADIANS)
if(MSVC)
  target_compile_options(wf_editor PUBLIC /utf-8 /W4 /permissive-)
endif()

if(WF_BUILD_STUDIO)
  if(NOT WIN32)
    message(FATAL_ERROR "Studio currently targets Windows. Other platforms are planned later.")
  endif()
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SDL_REVISION "SDL-3.2.28-7f3ae3d57459" CACHE STRING "Pinned SDL revision" FORCE)
  FetchContent_Declare(SDL3 URL https://github.com/libsdl-org/SDL/archive/7f3ae3d57459e59943a4ecfefc8f6277ec6bf540.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_Declare(imgui URL https://github.com/ocornut/imgui/archive/af84778eeaab75bee0b54420f3157cf3ee00618f.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_Declare(glad URL https://github.com/Dav1dde/glad/archive/73db193f853e2ee079bf3ca8a64aa2eaf6459043.tar.gz SOURCE_SUBDIR cmake DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(SDL3 imgui glad)
  glad_add_library(wf_glad STATIC REPRODUCIBLE API gl:core=4.5)

  add_library(wf_imgui STATIC
    ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl3.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
  target_include_directories(wf_imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
  target_link_libraries(wf_imgui PUBLIC SDL3::SDL3 opengl32)

  add_library(wf_renderer src/render/viewport_renderer.cpp src/render/gpu_texture.cpp)
  target_include_directories(wf_renderer PUBLIC src)
  target_link_libraries(wf_renderer PUBLIC wf_domain PRIVATE wf_glad)
  add_library(wf_platform src/platform/sdl_host.cpp src/platform/glb_files.cpp)
  target_include_directories(wf_platform PUBLIC src)
  target_link_libraries(wf_platform PUBLIC SDL3::SDL3 PRIVATE wf_glad wf_import)
  if(MSVC)
    # 此目标不继承 wf_editor，单独启用 UTF-8 以读取双语注释。
    # This target does not inherit wf_editor; read bilingual sources as UTF-8.
    target_compile_options(wf_platform PRIVATE /utf-8)
  endif()
  add_executable(worldforge-studio WIN32
    apps/studio/main.cpp src/studio/studio_app.cpp src/ui/studio_panels.cpp)
  target_link_libraries(worldforge-studio PRIVATE wf_editor wf_renderer wf_platform wf_imgui wf_glad wf_import)
  target_compile_definitions(worldforge-studio PRIVATE WF_VERSION="${PROJECT_VERSION}-texture-dev")
  install(TARGETS worldforge-studio RUNTIME DESTINATION .)
  install(FILES ${tinygltf_SOURCE_DIR}/attic/examples/common/LICENSE.lodepng DESTINATION licenses RENAME lodepng.txt)
  install(FILES ${tinygltf_SOURCE_DIR}/LICENSE DESTINATION licenses RENAME tinygltf.txt)
  install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
  install(DIRECTORY docs DESTINATION .)
  install(FILES ${SDL3_SOURCE_DIR}/LICENSE.txt DESTINATION licenses RENAME SDL.txt)
  install(FILES ${imgui_SOURCE_DIR}/LICENSE.txt DESTINATION licenses RENAME imgui.txt)
  install(FILES ${glm_SOURCE_DIR}/copying.txt DESTINATION licenses RENAME glm.txt)
  install(FILES ${glad_SOURCE_DIR}/LICENSE DESTINATION licenses RENAME glad.txt)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
  include(InstallRequiredSystemLibraries)
endif()

if(WF_BUILD_TESTS)
  enable_testing()
  if(DEFINED Python_EXECUTABLE AND NOT DEFINED Python3_EXECUTABLE)
    set(Python3_EXECUTABLE "${Python_EXECUTABLE}")
  endif()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_custom_target(wf_mesh_fixtures
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/generate_mesh_fixtures.py ${CMAKE_BINARY_DIR}/fixtures)
  add_executable(wf_import_tests tests/glb_importer_test.cpp)
  target_link_libraries(wf_import_tests PRIVATE wf_import)
  target_compile_definitions(wf_import_tests PRIVATE WF_FIXTURE_DIR="${CMAKE_BINARY_DIR}/fixtures")
  add_dependencies(wf_import_tests wf_mesh_fixtures)
  add_test(NAME glb_importer COMMAND wf_import_tests)
  add_executable(wf_import_service_tests tests/import_service_test.cpp)
  target_link_libraries(wf_import_service_tests PRIVATE wf_import wf_editor)
  target_compile_definitions(wf_import_service_tests PRIVATE WF_FIXTURE_DIR="${CMAKE_BINARY_DIR}/fixtures")
  add_dependencies(wf_import_service_tests wf_mesh_fixtures)
  add_test(NAME import_service COMMAND wf_import_service_tests)
  set_tests_properties(import_service PROPERTIES TIMEOUT 15)
  add_executable(wf_editor_tests tests/editor_state_test.cpp)
  target_link_libraries(wf_editor_tests PRIVATE wf_editor)
  add_test(NAME editor_state COMMAND wf_editor_tests)
  if(WF_BUILD_STUDIO)
    add_executable(wf_selection_ui_tests tests/selection_ui_test.cpp src/ui/studio_panels.cpp)
    target_link_libraries(wf_selection_ui_tests PRIVATE wf_editor wf_renderer wf_platform wf_imgui wf_glad)
    add_executable(wf_texture_gpu_tests tests/texture_gpu_test.cpp)
    target_link_libraries(wf_texture_gpu_tests PRIVATE wf_editor wf_renderer wf_platform wf_import wf_glad)
    target_compile_definitions(wf_texture_gpu_tests PRIVATE WF_FIXTURE_DIR="${CMAKE_BINARY_DIR}/fixtures")
    add_dependencies(wf_texture_gpu_tests wf_mesh_fixtures)
    add_executable(wf_mesh_gpu_tests tests/mesh_gpu_test.cpp)
    target_link_libraries(wf_mesh_gpu_tests PRIVATE wf_editor wf_renderer wf_platform wf_import wf_glad)
    target_compile_definitions(wf_mesh_gpu_tests PRIVATE WF_FIXTURE_DIR="${CMAKE_BINARY_DIR}/fixtures")
    add_dependencies(wf_mesh_gpu_tests wf_mesh_fixtures)
  endif()
  if(WF_BUILD_STUDIO AND WF_ENABLE_GPU_TESTS)
    add_test(NAME studio_gpu_smoke COMMAND worldforge-studio --smoke-test "${CMAKE_BINARY_DIR}/smoke")
    add_test(NAME studio_mesh_smoke COMMAND worldforge-studio --smoke-test "${CMAKE_BINARY_DIR}/mesh_smoke" "${CMAKE_BINARY_DIR}/fixtures/multiple_primitives.glb")
    set_tests_properties(studio_mesh_smoke PROPERTIES TIMEOUT 60 LABELS "gpu")
    add_test(NAME selection_ui COMMAND wf_selection_ui_tests)
    set_tests_properties(selection_ui PROPERTIES TIMEOUT 60 LABELS "gpu")
    add_test(NAME texture_gpu COMMAND wf_texture_gpu_tests)
    set_tests_properties(texture_gpu PROPERTIES TIMEOUT 60 LABELS "gpu")
    add_test(NAME mesh_gpu_smoke COMMAND wf_mesh_gpu_tests)
    set_tests_properties(mesh_gpu_smoke PROPERTIES TIMEOUT 60 LABELS "gpu")
    set_tests_properties(studio_gpu_smoke PROPERTIES TIMEOUT 60 LABELS "gpu")
  endif()
endif()
